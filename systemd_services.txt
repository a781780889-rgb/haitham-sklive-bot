# systemd Services للنظام المتكامل
# ضع هذه الملفات في: /etc/systemd/system/

# ═══════════════════════════════════════════════════════════
# 1. seha-api.service - API Server
# ═══════════════════════════════════════════════════════════

[Unit]
Description=Seha API Server
After=network.target

[Service]
Type=simple
User=www-data
WorkingDirectory=/path/to/enhanced_bot
Environment="PATH=/usr/bin:/usr/local/bin"
EnvironmentFile=/path/to/enhanced_bot/.env
ExecStart=/usr/bin/python3 /path/to/enhanced_bot/api_server.py
Restart=always
RestartSec=10

[Install]
WantedBy=multi-user.target

# ═══════════════════════════════════════════════════════════
# 2. seha-web.service - Web Server
# ═══════════════════════════════════════════════════════════

[Unit]

# تعليمات التثبيت:
# ═══════════════════════════════════════════════════════════

# 1. نسخ ملفات الخدمات:
#    sudo cp systemd_services.txt /etc/systemd/system/
#    (أو قم بنسخ كل قسم في ملف منفصل)

# 2. تحديث المسارات:
